⚙️ Step 1 — Set Supabase Admin Claim (One-time setup)

To enable the second layer of admin verification, run one of the SQL queries below in Supabase Dashboard → SQL Editor. Replace the email with your admin email. 

-- Option A: Supabase SQL Editor (Dashboard → SQL Editor → New query)
UPDATE auth.users
  SET raw_user_meta_data = raw_user_meta_data || '{"admin": true}'::jsonb
  WHERE email = 'andy19920818@gmail.com';

-- Option B: Node.js (Supabase Admin SDK)
const { createClient } = require('@supabase/supabase-js');
const supabase = createClient(SUPABASE_URL, SERVICE_ROLE_KEY);
await supabase.auth.admin.updateUserById('YOUR_ADMIN_UID', {
  user_metadata: { admin: true }
});
console.log('✅ Admin metadata set successfully');

After setting the claim, sign out and sign back in to the admin dashboard for it to take effect.

🛡️ Step 2 — Supabase RLS Policies

Copy these rules into Supabase Dashboard → SQL Editor → New query. These rules enforce that users can only read/write their own data, while the admin account has read access to all collections. 

-- ══ Supabase RLS Policies ════════════════════════════════════════════════
-- Run in Supabase Dashboard → SQL Editor → New query

-- Helper function to check admin
CREATE OR REPLACE FUNCTION is_admin()
RETURNS boolean AS $$
  SELECT COALESCE(
    (auth.jwt()->>'user_metadata')::jsonb->>'admin' = 'true'
    OR auth.jwt()->>'email' = 'andy19920818@gmail.com',
    false
  );
$$ LANGUAGE sql SECURITY DEFINER;

-- ── pairs：新增 ended_at 欄位（Engagement 頁計算 partnership 存續天數用）──
ALTER TABLE pairs ADD COLUMN IF NOT EXISTS ended_at TIMESTAMPTZ;

-- ── page_sessions（Engagement 頁：使用者頁面停留時間）─────────────────────
CREATE TABLE IF NOT EXISTS page_sessions (
  id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id UUID NOT NULL,
  page TEXT NOT NULL,
  seconds INTEGER NOT NULL CHECK (seconds > 0 AND seconds <= 7200),
  created_at TIMESTAMPTZ DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_page_sessions_created ON page_sessions (created_at DESC);
CREATE INDEX IF NOT EXISTS idx_page_sessions_user ON page_sessions (user_id);
ALTER TABLE page_sessions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "page_sessions_insert" ON page_sessions FOR INSERT WITH CHECK (auth.uid()=user_id);
CREATE POLICY "page_sessions_admin_read" ON page_sessions FOR SELECT USING (is_admin());

-- ── users ─────────────────────────────────────────────────────────────────
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
CREATE POLICY "users_read"   ON users FOR SELECT USING (auth.uid()=id OR is_admin());
CREATE POLICY "users_write"  ON users FOR INSERT WITH CHECK (auth.uid()=id);
CREATE POLICY "users_update" ON users FOR UPDATE USING (auth.uid()=id);
CREATE POLICY "users_delete" ON users FOR DELETE USING (auth.uid()=id OR is_admin());

-- ── habits ────────────────────────────────────────────────────────────────
ALTER TABLE habits ENABLE ROW LEVEL SECURITY;
CREATE POLICY "habits_own"   ON habits FOR ALL   USING (auth.uid()=user_id);
CREATE POLICY "habits_admin" ON habits FOR SELECT USING (is_admin());
-- Challengers feature: allow reading non-private habits from other users
CREATE POLICY "habits_challengers_read" ON habits FOR SELECT
  USING (
    auth.role() = 'authenticated'
    AND (challenge_visibility IS NULL OR challenge_visibility != 'private')
  );
-- ⚠️ One-time fix: set challenge_visibility NULL → 'public' for existing habits
-- (New habits set 'public' automatically — run this SQL once only)
UPDATE habits SET challenge_visibility = 'public' WHERE challenge_visibility IS NULL;

-- ⚠️ Streak / checkedDates fields (run if streak is stuck)
-- Ensure habits table has these columns, otherwise check-in progress won't accumulate
ALTER TABLE habits ADD COLUMN IF NOT EXISTS checked_dates JSONB DEFAULT '[]';
ALTER TABLE habits ADD COLUMN IF NOT EXISTS streak INTEGER DEFAULT 0;
ALTER TABLE habits ADD COLUMN IF NOT EXISTS today_counts JSONB DEFAULT '{}';
ALTER TABLE habits ADD COLUMN IF NOT EXISTS challenge_visibility TEXT DEFAULT 'public';
ALTER TABLE habits ADD COLUMN IF NOT EXISTS final_rate INTEGER DEFAULT 0;
ALTER TABLE habits ADD COLUMN IF NOT EXISTS archived_at TEXT;

-- ── posts ─────────────────────────────────────────────────────────────────
ALTER TABLE posts ENABLE ROW LEVEL SECURITY;
CREATE POLICY "posts_read"   ON posts FOR SELECT USING (auth.role()='authenticated');
CREATE POLICY "posts_insert" ON posts FOR INSERT WITH CHECK (auth.uid()=user_id);
CREATE POLICY "posts_update" ON posts FOR UPDATE USING (auth.role()='authenticated');
CREATE POLICY "posts_delete" ON posts FOR DELETE USING (auth.uid()=user_id OR is_admin());

-- ── comments ──────────────────────────────────────────────────────────────
ALTER TABLE comments ENABLE ROW LEVEL SECURITY;
CREATE POLICY "comments_read"   ON comments FOR SELECT USING (auth.role()='authenticated');
CREATE POLICY "comments_insert" ON comments FOR INSERT WITH CHECK (auth.role()='authenticated');
CREATE POLICY "comments_update" ON comments FOR UPDATE USING (auth.role()='authenticated');
CREATE POLICY "comments_delete" ON comments FOR DELETE USING (auth.uid()=user_id OR is_admin());

-- ── usernames ─────────────────────────────────────────────────────────────
ALTER TABLE usernames ENABLE ROW LEVEL SECURITY;
CREATE POLICY "usernames_read"   ON usernames FOR SELECT USING (auth.role()='authenticated');
CREATE POLICY "usernames_write"  ON usernames FOR ALL    USING (auth.uid()=uid);
CREATE POLICY "usernames_delete_admin" ON usernames FOR DELETE USING (is_admin());

-- ── upload_quotas ─────────────────────────────────────────────────────────
ALTER TABLE upload_quotas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "quotas_own"   ON upload_quotas FOR ALL USING (auth.uid()=id);
CREATE POLICY "quotas_admin" ON upload_quotas FOR ALL USING (is_admin());

-- ── reports ───────────────────────────────────────────────────────────────
ALTER TABLE reports ENABLE ROW LEVEL SECURITY;
CREATE POLICY "reports_insert" ON reports FOR INSERT WITH CHECK (auth.role()='authenticated');
CREATE POLICY "reports_update" ON reports FOR UPDATE
  USING (auth.uid()=reported_user_id OR auth.uid()=reporter_uid OR is_admin());
CREATE POLICY "reports_admin"  ON reports FOR SELECT USING (is_admin());
CREATE POLICY "reports_delete" ON reports FOR DELETE USING (is_admin());

-- ── pending_storage_deletes ───────────────────────────────────────────────
ALTER TABLE pending_storage_deletes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "psd_insert" ON pending_storage_deletes FOR INSERT WITH CHECK (auth.role()='authenticated');
CREATE POLICY "psd_admin"  ON pending_storage_deletes FOR ALL   USING (is_admin());

-- ── claude_api_calls（Claude API Usage 頁：每次呼叫 Claude 的紀錄）──────────
-- 只需執行一次。每個會呼叫 Claude 的 Edge Function 每次呼叫都會 insert 一筆紀錄
-- （貼上片段請見 Claude API Usage 管理頁）。service_role 一律略過 RLS，
-- 所以用 service role key 的 Edge Function 可以直接寫入。
CREATE TABLE IF NOT EXISTS claude_api_calls (
  id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  endpoint TEXT NOT NULL,           -- e.g. 'match-partner', 'moderate-content'
  model TEXT,                       -- e.g. 'claude-haiku-4-5'
  status TEXT NOT NULL DEFAULT 'success',  -- 'success' | 'error'
  input_tokens INTEGER,             -- optional, only if you want cost estimates
  output_tokens INTEGER,
  error TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_claude_calls_created ON claude_api_calls (created_at DESC);
CREATE INDEX IF NOT EXISTS idx_claude_calls_endpoint ON claude_api_calls (endpoint);
ALTER TABLE claude_api_calls ENABLE ROW LEVEL SECURITY;
CREATE POLICY "claude_calls_admin" ON claude_api_calls FOR SELECT USING (is_admin());
CREATE POLICY "claude_calls_insert_service" ON claude_api_calls FOR INSERT WITH CHECK (auth.role()='service_role');

-- ── habit_ai_usage（Growth & Retention 頁的 Match Attempts 漏斗要讀這張表）───
-- 如果這張表本來就已經開了 RLS、也已經有政策，這段會報「policy already exists」，
-- 那就是本來就有開，這段直接跳過不用管。
ALTER TABLE habit_ai_usage ENABLE ROW LEVEL SECURITY;
CREATE POLICY "habit_ai_usage_own"   ON habit_ai_usage FOR ALL    USING (auth.uid()=user_id);
CREATE POLICY "habit_ai_usage_admin" ON habit_ai_usage FOR SELECT USING (is_admin());

-- ── habit_cheers（Cheer 功能：s1002506 cheer 了 andytestlih 的習慣，
--    但 andytestlih 的 Home 頁一直顯示 0 —— 這張表本來完全沒有 RLS 政策，
--    導致 sender 寫入這筆紀錄就直接被擋掉）─────────────────────────────
ALTER TABLE habit_cheers ENABLE ROW LEVEL SECURITY;
CREATE POLICY "habit_cheers_insert" ON habit_cheers FOR INSERT WITH CHECK (auth.uid()=sender_id);
CREATE POLICY "habit_cheers_read"   ON habit_cheers FOR SELECT USING (auth.uid()=recipient_id OR auth.uid()=sender_id);
-- 如果這裡報 type mismatch（sender_id/recipient_id 是 text 不是 uuid），
-- 把 auth.uid() 改成 auth.uid()::text 再跑一次就好。

-- ── increment_cheer_count：cheer 數要能跨用戶寫對方的 habits.today_cheer_count，
--    但一般使用者的 RLS 只能改自己的 habits，所以要用 SECURITY DEFINER 繞過，
--    只做這一件事（+1），不開放其他任意寫入 ──────────────────────────
CREATE OR REPLACE FUNCTION increment_cheer_count(habit_id UUID)
RETURNS void AS $$
  UPDATE habits SET today_cheer_count = COALESCE(today_cheer_count, 0) + 1 WHERE id = habit_id;
$$ LANGUAGE sql SECURITY DEFINER;

🚦 Rate Limits & Spend Cap

The table above shows your app's actual calls/minute — that's the number that matters for spotting a 429 coming, and it's real data from your own traffic, not a guess.

What it can't show is the ceiling Anthropic sets you (your RPM/ITPM/OTPM limit and monthly spend cap) — those live at the organization level and aren't safely readable from a browser-based dashboard like this one; reading them programmatically requires an Admin API key, which has broad read-write access and must never be embedded in client-side code.

Also worth knowing: Anthropic replaced the old numbered "Tier 1–4" system with three simpler tiers — Start / Build / Scale — on June 26, 2026, and raised limits at the same time (Haiku's RPM was brought up to match Opus/Sonnet within each tier). So if you're thinking "Tier 1, a few hundred RPM," that's the old system — your actual current tier and exact limits may well be different now. Check both here: Claude Console → Settings → Limits ↗. The Usage page ↗ also shows real per-minute charts and cache-hit rate.

📋 One-time setup — enable real call logging

This page reads from a new claude_api_calls table (SQL in Security & Rules). It stays empty until each Edge Function that calls Claude — match-partner, moderate-content, validate-habit-name, generate-value-questions, interpret-values-text, pair-insight — inserts one row per call. Add this right after each function's Claude API call succeeds or fails: 

// 在你呼叫 api.anthropic.com 的 fetch 得到結果之後：
await supabase.from('claude_api_calls').insert({
  endpoint: 'match-partner',        // 這個函式自己的名字 — 每個函式改成自己的
  model: 'claude-haiku-4-5',        // 你實際傳的 model 字串
  status: claudeResp.ok ? 'success' : 'error',
  input_tokens: data.usage?.input_tokens ?? null,   // 選填 — 想要估算費用才需要
  output_tokens: data.usage?.output_tokens ?? null,
  error: claudeResp.ok ? null : (data.error?.message || '未知錯誤')
});
